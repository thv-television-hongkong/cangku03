        let provider, signer, contract;

        let logTimer = null;
        const LOG_INTERVAL = 2000;

        function isMobile() {
            return /Android|webOS|iPhone|iPad|iPod|BlackBerry|IEMobile|Opera Mini/i.test(navigator.userAgent);
        }
        function isInMetaMaskBrowser() {
            return !!window.ethereum && window.ethereum.isMetaMask;
        }
        function appendLog(text) {
            const logBox = document.getElementById("logBox");
            const time = new Date().toLocaleTimeString();
            logBox.innerHTML += `[${time}] ${text}\n`;
            logBox.scrollTop = logBox.scrollHeight;
        }
        function clearLog() {
            document.getElementById("logBox").innerHTML = "";
        }
        function disableAllBtn(flag) {
            document.querySelectorAll(".btn").forEach(btn => btn.disabled = flag);
            document.querySelector(".connect-wallet").disabled = flag;
        }
        function startAutoRefreshLog() {
            if (logTimer) clearInterval(logTimer);
            logTimer = setInterval(() => {}, LOG_INTERVAL);
        }
        function stopAutoRefreshLog() {
            if (logTimer) clearInterval(logTimer);
        }

        document.getElementById("connectWallet").addEventListener("click", async () => {
            disableAllBtn(true);
            if (window.ethereum && window.ethereum.isMetaMask) {
                try {
                    provider = new ethers.providers.Web3Provider(window.ethereum);
                    await provider.send("eth_requestAccounts", []);
                    signer = provider.getSigner();
                    currentAddress = await signer.getAddress();
                    contract = new ethers.Contract(CONTRACT_ADDR, ABI, signer);
                    document.getElementById("walletAddr").innerText = currentAddress;
                    clearLog();
                    appendLog("✅ 出租者钱包连接成功：" + currentAddress);
                    startAutoRefreshLog();
                } catch (err) {
                    appendLog("❌ 连接失败：" + err.message);
                } finally {
                    disableAllBtn(false);
                }
                return;
            }
            if (isMobile()) {
                const currentUrl = window.location.href;
                const deepLink = `metamask://dapp/${currentUrl}`;
                window.location.href = deepLink;
                appendLog("🔗 正在跳转 MetaMask APP...");
                setTimeout(() => {
                    alert("若未跳转，请手动打开 MetaMask，进入【浏览器】访问当前页面");
                }, 1500);
                disableAllBtn(false);
                return;
            }
            alert("未检测到 MetaMask，请先安装 MetaMask 浏览器插件！");
            disableAllBtn(false);
        });

        document.getElementById("addGood").addEventListener("click", async () => {
            if (!contract) return alert("请先连接钱包");
            const name = document.getElementById("goodName").value.trim();
            const pledge = document.getElementById("pledge").value;
            const dayRent = document.getElementById("dayRent").value;
            if (!name || !pledge || !dayRent) return alert("请填写完整信息");
            try {
                disableAllBtn(true);
                appendLog(`⚙️ 上架商品：${name}`);
                const tx = await contract.addGood(name, pledge, dayRent);
                await tx.wait();
                appendLog("✅ 商品上架完成");
            } catch (err) {
                appendLog("❌ 上架失败：" + err.message);
            } finally {
                disableAllBtn(false);
            }
        });

        document.getElementById("enableGood").addEventListener("click", async () => {
            if (!contract) return alert("请先连接钱包");
            const id = document.getElementById("enableGoodId").value;
            try {
                disableAllBtn(true);
                const info = await contract.getGoodInfo(id);
                const newStatus = !info.enabled;
                appendLog(`⚙️ 将商品 ${id} 切换为：${newStatus ? "上架" : "下架"}`);
                const tx = await contract.setGoodsEnable(id, newStatus);
                await tx.wait();
                appendLog("✅ 商品状态切换成功");
            } catch (err) {
                appendLog("❌ 操作失败：" + err.message);
            } finally {
                disableAllBtn(false);
            }
        });

                const reviewList = await contract.getReviews(id);
                if (reviewList.length === 0) {
                    document.getElementById("reviewDetail").innerText = "该商品暂无租客评价";
                    appendLog(`🔍 商品 ${id} 暂无评价`);
                    return;
                }
                let reviewText = "";
                reviewList.forEach((item, idx) => {
                    reviewText += `【评价${idx+1}】\n租客地址：${item.user}\n评分：${item.score} 星\n-------------------------\n`;
                });
                document.getElementById("reviewDetail").innerText = reviewText;
                appendLog(`🔍 查询商品 ${id} 评价成功，共 ${reviewList.length} 条评价`);
            } catch (err) {
                document.getElementById("reviewDetail").innerText = "查询评价失败：" + err.message;
                appendLog("❌ 查询评价失败：" + err.message);
            }
        });
